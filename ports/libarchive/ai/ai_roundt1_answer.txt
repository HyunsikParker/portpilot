+qnx_fstatvfs(int fd, struct statvfs *st)
+{
+	int r, nfd, e;
+
+	r = fstatvfs(fd, st);
+	if (r == -1 && (errno == ENOSYS || errno == EINVAL)) {
+		nfd = openat(fd, ".", O_RDONLY | O_CLOEXEC);
+		if (nfd >= 0) {
+			r = fstatvfs(nfd, st);
+			e = errno;
+			close(nfd);
+			errno = e;
+		} else
+			errno = ENOSYS;
+	}
+	return (r);
+}
